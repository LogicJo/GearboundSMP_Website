/*
========================================================
GITHUB CONFIGURATION
========================================================
*/


const GITHUB_OWNER =
    "LogicJo";


const GITHUB_REPOSITORY =
    "GearboundSMP_Website";


const GITHUB_BRANCH =
    "master";


/*
 * IMPORTANT:
 *
 * This is the folder containing the JSON files.
 *
 * It is NOT the "playerstats" website folder.
 *
 * Example:
 *
 * /playerstats/
 *     index.html
 *
 * /Playerstats/
 *     Player1.json
 *     Player2.json
 *
 */


const PLAYERSTATS_FOLDER =
    "Playerstats";


/*
========================================================
GITHUB API URL
========================================================
*/


const GITHUB_API_URL =
    `https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPOSITORY}/contents/${PLAYERSTATS_FOLDER}?ref=${GITHUB_BRANCH}`;



/*
========================================================
LOAD PLAYER STATS
========================================================
*/


async function loadPlayerStats(){


    const loading =
        document.getElementById("loading");


    const grid =
        document.getElementById("playerGrid");


    const error =
        document.getElementById("error");


    const errorMessage =
        document.getElementById("errorMessage");


    const searchInput =
        document.getElementById("searchInput");


    const search =
        searchInput.value
            .trim()
            .toLowerCase();


    /*
    Reset page
    */

    loading.classList.remove("hidden");

    error.classList.add("hidden");

    grid.innerHTML = "";


    try{


        /*
        ================================================
        GET CONTENTS OF PLAYERSTATS FOLDER
        ================================================
        */


        const response =
            await fetch(GITHUB_API_URL, {

                headers: {

                    "Accept":
                        "application/vnd.github+json"

                }

            });


        if(!response.ok){

            throw new Error(
                `GitHub API returned HTTP ${response.status}`
            );

        }


        const files =
            await response.json();


        /*
        ================================================
        FIND JSON FILES
        ================================================
        */


        const jsonFiles =
            files.filter(file =>

                file.type === "file" &&

                file.name
                    .toLowerCase()
                    .endsWith(".json")

            );


        /*
        Update JSON file counter
        */

        document.getElementById("fileCount")
            .innerText =
                jsonFiles.length;


        /*
        ================================================
        LOAD ALL JSON FILES
        ================================================
        */


        const players = [];


        for(const file of jsonFiles){


            try{


                const jsonResponse =
                    await fetch(file.download_url);


                if(!jsonResponse.ok){

                    console.warn(
                        "Could not load:",
                        file.name
                    );

                    continue;

                }


                const data =
                    await jsonResponse.json();


                players.push({

                    fileName:
                        file.name,

                    data:
                        data

                });


            }

            catch(fileError){

                console.warn(
                    "Invalid JSON:",
                    file.name,
                    fileError
                );

            }

        }


        /*
        ================================================
        SORT PLAYERS
        ================================================
        */


        players.sort((a,b) =>

            getPlayerName(
                a.data,
                a.fileName
            ).localeCompare(

                getPlayerName(
                    b.data,
                    b.fileName
                )

            )

        );


        /*
        ================================================
        SEARCH
        ================================================
        */


        const filteredPlayers =
            players.filter(player => {


                const name =
                    getPlayerName(
                        player.data,
                        player.fileName
                    );


                return name
                    .toLowerCase()
                    .includes(search);

            });


        /*
        ================================================
        UPDATE PLAYER COUNT
        ================================================
        */


        document.getElementById("playerCount")
            .innerText =
                filteredPlayers.length;


        /*
        ================================================
        UPDATE LAST UPDATE
        ================================================
        */


        document.getElementById("lastUpdate")
            .innerText =
                new Date().toLocaleString(
                    "en-GB",
                    {
                        day:"2-digit",
                        month:"2-digit",
                        year:"numeric",
                        hour:"2-digit",
                        minute:"2-digit"
                    }
                );


        /*
        ================================================
        CREATE PLAYER CARDS
        ================================================
        */


        filteredPlayers.forEach(player => {


            createPlayerCard(

                player.data,

                player.fileName

            );


        });


        /*
        ================================================
        NO RESULTS
        ================================================
        */


        if(filteredPlayers.length === 0){


            grid.innerHTML = `

                <div
                    class="launcher-panel p-10 text-center col-span-full">


                    <div
                        class="text-5xl mb-5">

                        🔎

                    </div>


                    <div
                        class="text-2xl font-bold">

                        No players found

                    </div>


                    <div
                        class="text-zinc-500 mt-3">

                        Try searching for another player.

                    </div>


                </div>

            `;


        }


    }


    catch(e){


        console.error(e);


        errorMessage.innerText =
            e.message;


        error.classList.remove("hidden");


    }


    finally{


        loading.classList.add("hidden");


    }

}



/*
========================================================
GET PLAYER NAME
========================================================
*/


function getPlayerName(data,fileName){


    /*
    Try common property names first
    */


    if(typeof data === "object" && data !== null){


        if(data.name)
            return String(data.name);


        if(data.username)
            return String(data.username);


        if(data.player)
            return String(data.player);


        if(data.playerName)
            return String(data.playerName);


        if(data.displayName)
            return String(data.displayName);


    }


    /*
    Fallback:
    Use filename
    */


    return fileName
        .replace(/\.json$/i,"");

}

/*
========================================================
FIND VALUE
========================================================
*/


function findValue(data,possibleNames,fallback){


    if(
        typeof data !== "object" ||
        data === null
    ){

        return fallback;

    }


    /*
    Direct properties
    */


    for(const name of possibleNames){


        if(
            Object.prototype.hasOwnProperty.call(
                data,
                name
            )
        ){

            return data[name];

        }

    }


    /*
    Look inside common statistics objects
    */


    const containers = [

        data.statistics,

        data.stats,

        data.playerStats,

        data.data

    ];


    for(const container of containers){


        if(
            typeof container !== "object" ||
            container === null
        ){

            continue;

        }


        for(const name of possibleNames){


            if(
                Object.prototype.hasOwnProperty.call(
                    container,
                    name
                )
            ){

                return container[name];

            }

        }

    }


    return fallback;

}



/*
========================================================
FORMAT VALUES
========================================================
*/


function formatValue(value){


    if(value === null ||
       value === undefined){

        return "—";

    }


    if(typeof value === "object"){

        return JSON.stringify(value);

    }


    return String(value);

}



/*
========================================================
HTML ESCAPE
========================================================
*/


function escapeHTML(value){


    return String(value)

        .replaceAll("&","&amp;")

        .replaceAll("<","&lt;")

        .replaceAll(">","&gt;")

        .replaceAll('"',"&quot;")

        .replaceAll("'","&#039;");

}
